# Examples

## curl

```bash
# Get a single tweet
curl http://localhost:8000/posts/20

# Get a user's profile
curl http://localhost:8000/profile/elonmusk

# Get a user's timeline (limit 5)
curl "http://localhost:8000/posts/user/elonmusk?limit=5"

# Search tweets
curl "http://localhost:8000/search?q=openai&limit=10"

# Advanced search with filters
curl "http://localhost:8000/search/advanced?q=ai&from_user=elonmusk&min_faves=1000&since=2024-01-01"

# Get replies to a tweet
curl http://localhost:8000/replies/20

# Get quote tweets
curl http://localhost:8000/quotes/20

# Get current trends
curl http://localhost:8000/trends/global

# Get media from a tweet
curl http://localhost:8000/media/tweet/20

# Bulk fetch tweets
curl "http://localhost:8000/posts/ids?ids=20,21,22"

# Check provider health
curl http://localhost:8000/providers/health
```

## Python (httpx)

```python
import httpx

BASE_URL = "http://localhost:8000"

# Get a tweet
r = httpx.get(f"{BASE_URL}/posts/20")
tweet = r.json()["data"]
print(f"@{tweet['author']['username']}: {tweet['text']}")
print(f"Likes: {tweet['like_count']:,}")

# Get a profile
r = httpx.get(f"{BASE_URL}/profile/elonmusk")
profile = r.json()["data"]
print(f"{profile['display_name']} (@{profile['username']})")
print(f"Followers: {profile['followers']:,}")

# Search
r = httpx.get(f"{BASE_URL}/search", params={"q": "openai", "limit": 10})
for tweet in r.json()["data"]:
    print(f"@{tweet['author']['username']}: {tweet['text'][:80]}...")

# Get user timeline with pagination
cursor = None
all_tweets = []
while len(all_tweets) < 100:
    params = {"limit": 20}
    if cursor:
        params["cursor"] = cursor
    r = httpx.get(f"{BASE_URL}/posts/user/elonmusk", params=params)
    data = r.json()
    all_tweets.extend(data["data"])
    cursor = data.get("next_cursor")
    if not cursor:
        break

print(f"Fetched {len(all_tweets)} tweets")
```

## Python (asyncio + httpx)

```python
import asyncio
import httpx

async def main():
    async with httpx.AsyncClient(base_url="http://localhost:8000") as client:
        # Parallel requests
        tweet_task = client.get("/posts/20")
        profile_task = client.get("/profile/elonmusk")
        trends_task = client.get("/trends/global")
        
        tweet_resp, profile_resp, trends_resp = await asyncio.gather(
            tweet_task, profile_task, trends_task
        )
        
        print(f"Tweet: {tweet_resp.json()['data']['text']}")
        print(f"Profile: {profile_resp.json()['data']['display_name']}")
        print(f"Top trend: {trends_resp.json()['trends'][0]['name']}")

asyncio.run(main())
```

## JavaScript (fetch)

```javascript
const BASE_URL = "http://localhost:8000";

// Get a tweet
const r = await fetch(`${BASE_URL}/posts/20`);
const { data: tweet, provider } = await r.json();
console.log(`@${tweet.author.username}: ${tweet.text}`);
console.log(`Likes: ${tweet.like_count.toLocaleString()}`);
console.log(`Served by: ${provider.name} in ${Math.round(provider.latency_ms)}ms`);

// Get a profile
const r2 = await fetch(`${BASE_URL}/profile/elonmusk`);
const { data: profile } = await r2.json();
console.log(`${profile.display_name} (@${profile.username})`);
console.log(`Followers: ${profile.followers.toLocaleString()}`);

// Search with pagination
async function searchAll(query, max = 100) {
  const results = [];
  let cursor = null;
  while (results.length < max) {
    const url = new URL(`${BASE_URL}/search`);
    url.searchParams.set("q", query);
    url.searchParams.set("limit", "20");
    if (cursor) url.searchParams.set("cursor", cursor);
    const r = await fetch(url);
    const { data, next_cursor } = await r.json();
    results.push(...data);
    if (!next_cursor) break;
    cursor = next_cursor;
  }
  return results.slice(0, max);
}

const tweets = await searchAll("openai", 50);
console.log(`Found ${tweets.length} tweets`);
```

## TypeScript

```typescript
import { Tweet, Profile, ProviderInfo, TweetResponse, ProfileResponse } from "blitzyxsc";

const BASE_URL = "http://localhost:8000";

async function getTweet(id: string): Promise<{ data: Tweet; provider: ProviderInfo }> {
  const r = await fetch(`${BASE_URL}/posts/${id}`);
  if (!r.ok) throw new Error(`Failed: ${r.status}`);
  return r.json();
}

async function getProfile(username: string): Promise<{ data: Profile; provider: ProviderInfo }> {
  const r = await fetch(`${BASE_URL}/profile/${username}`);
  if (!r.ok) throw new Error(`Failed: ${r.status}`);
  return r.json();
}

const { data: tweet, provider } = await getTweet("20");
console.log(`@${tweet.author?.username}: ${tweet.text}`);
console.log(`Served by ${provider.name} in ${provider.latency_ms}ms`);
```

## Go

```go
package main

import (
    "encoding/json"
    "fmt"
    "io"
    "net/http"
)

const baseURL = "http://localhost:8000"

type TweetResponse struct {
    Data     Tweet         `json:"data"`
    Provider ProviderInfo  `json:"provider"`
}

type Tweet struct {
    ID         string `json:"id"`
    Text       string `json:"text"`
    Author     Author `json:"author"`
    LikeCount  int    `json:"like_count"`
}

type Author struct {
    Username    string `json:"username"`
    DisplayName string `json:"display_name"`
}

type ProviderInfo struct {
    Name      string  `json:"name"`
    Ok        bool    `json:"ok"`
    LatencyMs float64 `json:"latency_ms"`
}

func getTweet(id string) (*TweetResponse, error) {
    resp, err := http.Get(fmt.Sprintf("%s/posts/%s", baseURL, id))
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()
    body, _ := io.ReadAll(resp.Body)
    var result TweetResponse
    if err := json.Unmarshal(body, &result); err != nil {
        return nil, err
    }
    return &result, nil
}

func main() {
    result, err := getTweet("20")
    if err != nil {
        panic(err)
    }
    fmt.Printf("@%s: %s\n", result.Data.Author.Username, result.Data.Text)
    fmt.Printf("Likes: %d\n", result.Data.LikeCount)
    fmt.Printf("Served by: %s in %.0fms\n", result.Provider.Name, result.Provider.LatencyMs)
}
```

## Error handling

All implementations return standard HTTP status codes:

| Status | Meaning |
|--------|---------|
| 200 | Success |
| 400 | Bad request (invalid params, malformed date, etc.) |
| 404 | Resource not found (tweet/profile doesn't exist) |
| 422 | Validation error (FastAPI auto-generates these) |
| 502 | All providers failed (upstream sources all down or errored) |

Error responses have this shape:

```json
{
  "detail": "all providers failed (composable,syndication,nitter,twscrape,snscrape,rss); last error: rss does not support get_replies"
}
```

In JavaScript:

```javascript
const r = await fetch(`${BASE_URL}/posts/${id}`);
if (r.status === 502) {
  const err = await r.json();
  console.error("All providers failed:", err.detail);
} else if (!r.ok) {
  throw new Error(`HTTP ${r.status}`);
} else {
  const { data } = await r.json();
  // use data
}
```
